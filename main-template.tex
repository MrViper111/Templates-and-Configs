    \par\medskip
    \ignorespacesafterend

% Usage: \diff[order]{f}[point]{x}
\NewDocumentCommand{\diff}{ o m o m }{%
    \IfNoValueTF{#1}{%
        \IfNoValueTF{#3}{%
            \frac{\mathrm{d}#2}{\mathrm{d}#4}
        }{%
            \frac{\mathrm{d}#2(#3)}{\mathrm{d}#4}
        }%
    }{%
        \frac{\mathrm{d}^{#1}#2}{\mathrm{d}#4^{#1}}
    }%
}

% Usage: \pdiff[order]{f}[point]{x}
\NewDocumentCommand{\pdiff}{ o m o m }{%
    \IfNoValueTF{#1}{%
        \IfNoValueTF{#3}{%
            \frac{\partial #2}{\partial #4}
        }{%
            \frac{\partial #2(#3)}{\partial #4}
        }%
    }{%
        \frac{\partial^{#1}#2}{\partial #4^{#1}}
    }%
}
